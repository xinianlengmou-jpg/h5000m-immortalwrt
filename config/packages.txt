# 插件清单。要加插件就往这个文件追加一行 CONFIG_PACKAGE_xxx=y。
# 当前只有「保底项」——5G 模组拨号与模组管理界面，刷完必须能上网、能进界面。
# 其它插件（文件共享、科学上网、内网穿透、Web 终端…）后面再一个个加。

# ---------- 5G 模组：拨号核心 ----------
# input-support 是 input/HID/USB 串口内核模块的 Kconfig 门控包。
# 没开的话 kmod-usb-acm / kmod-usb-serial-qualcomm 整条链被静默隐藏，
# qmodem 及其 select 的组件全都进不了固件（包写了 =y 也没用）。
CONFIG_PACKAGE_input-support=y

CONFIG_PACKAGE_qmodem=y
CONFIG_PACKAGE_quectel-CM-5G-M=y

# QMI 驱动 choice 挂在 qmodem 主包上（符号为 PACKAGE_qmodem_INCLUDE_*）。
# 显式选通用 QMI，避免厂商 qmi_wwan_q/f/s 与通用驱动同名模块冲突。
CONFIG_PACKAGE_qmodem_INCLUDE_generic-qmi-wwan=y
# 拨号 CM choice 显式声明，防止上游改默认值后静默漂移
CONFIG_PACKAGE_qmodem_USE_TOM_CUSTOMIZED_QUECTEL_CM=y

# ---------- 5G 模组：USB 基础 ----------
CONFIG_PACKAGE_kmod-usb2=y
CONFIG_PACKAGE_kmod-usb3=y
CONFIG_PACKAGE_kmod-usb-serial=y
CONFIG_PACKAGE_kmod-usb-serial-option=y
CONFIG_PACKAGE_kmod-usb-net-qmi-wwan=y

# ---------- 模组管理界面 ----------
# 纯 qmodem ubus 链路的通用美化 UI，不绑定模组型号。
# 自带 extra_modem_support.json（含 Quectel RG520N-CN），
# 安装时与每次开机自动幂等注入 QModem 支持库，无需手工改 modem_support.json。
CONFIG_PACKAGE_luci-app-qmodem-generic=y
CONFIG_PACKAGE_luci-i18n-qmodem-generic-zh-cn=y

# 两个旧前端不装，避免菜单路径与 ACL 条目冲突
CONFIG_PACKAGE_luci-app-qmodem=n
CONFIG_PACKAGE_luci-app-qmodem-next=n
CONFIG_PACKAGE_luci-i18n-qmodem-zh-cn=n

#  下面留空：以后要加的插件往这里追加
# ============================================================
